```python
from datetime import datetime, timedelta
import math

def penalty(reported: datetime, arrived: datetime, monthly_fee: float) -> float:
    """
    Calculate the penalty in euros for a late arrival.

    Parameters
    ----------
    reported : datetime
        The datetime when the service was reported.
    arrived : datetime
        The datetime when the service actually arrived.
    monthly_fee : float
        The monthly fee in euros.

    Returns
    -------
    float
        The penalty in euros. 0 if the arrival is within 48 hours of the report.
    """
    # Time difference between report and arrival
    delta = arrived - reported

    # If arrival is within 48 hours, no penalty
    if delta <= timedelta(hours=48):
        return 0.0

    # Calculate full days of delay (partial days count as a full day)
    days_of_delay = math.ceil(delta.total_seconds() / 86400)

    # 2% of the monthly fee per full day of delay
    penalty_amount = days_of_delay * 0.02 * monthly_fee
    return penalty_amount
```
