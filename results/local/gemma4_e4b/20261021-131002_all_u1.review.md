from datetime import datetime, timedelta

def penalty(reported: datetime, arrived: datetime, monthly_fee: float) -> float:
    """
    Calculates the penalty in euros based on the delay.

    Penalty is 2% of the fee per full day of delay beyond 48 hours (partial days count as a full day), 
    0 if within 48 hours.

    Args:
        reported: The reported arrival datetime.
        arrived: The actual arrival datetime.
        monthly_fee: The base monthly fee.

    Returns:
        The calculated penalty in euros.
    """
    delay = arrived - reported
    
    # 48 hours in timedelta
    threshold = timedelta(hours=48)
    
    if delay <= threshold:
        return 0.0
    else:
        # Calculate the delay exceeding 48 hours
        excess_delay = delay - threshold
        
        # Convert excess delay to total days (as a float)
        total_days_excess = excess_delay.total_seconds() / (24 * 3600)
        
        # Since partial days count as a full day, we use ceil to find the number of full days
        import math
        full_days_delay = math.ceil(total_days_excess)
        
        # Penalty is 2% (0.02) of the fee per full day
        penalty_amount = full_days_delay * 0.02 * monthly_fee
        return penalty_amount

# Example Calls
# Example 1: Delay of 72 hours (3 days total, 24 hours excess)
# 72 hours > 48 hours. Excess = 24 hours = 1 full day. Penalty = 1 * 0.02 * 100 = 2.0
reported1 = datetime(2023, 10, 1, 10, 0, 0)
arrived1 = datetime(2023, 10, 4, 10, 0, 0) # 3 days later
fee1 = 100.0
